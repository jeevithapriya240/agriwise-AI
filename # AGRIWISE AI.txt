crops = {
    "Tomato": {
        "demand": 90,
        "price": 85,
        "cost": 45000,
        "risk": 35
    },
    "Onion": {
        "demand": 82,
        "price": 75,
        "cost": 40000,
        "risk": 25
    },
    "Maize": {
        "demand": 75,
        "price": 65,
        "cost": 30000,
        "risk": 20
    },
    "Groundnut": {
        "demand": 70,
        "price": 72,
        "cost": 35000,
        "risk": 18
    }
}


def calculate(crop, land, transport, budget):

    data = crops[crop]

    cultivation_cost = data["cost"] * land
    total_cost = cultivation_cost + transport

    revenue = data["price"] * 1000 * land
    profit = revenue - total_cost

    budget_score = 100 if total_cost <= budget else 40

    score = (
        data["demand"] * 0.40
        + data["price"] * 0.30
        + (100 - data["risk"]) * 0.20
        + budget_score * 0.10
    )

    return {
        "demand": data["demand"],
        "price": data["price"],
        "cost": total_cost,
        "revenue": revenue,
        "profit": profit,
        "risk": data["risk"],
        "score": round(score)
    }


print("=" * 55)
print("             🌾 AGRIWISE AI 🌾")

print("=" * 55)

location = input("\n📍 Enter your location: ")
land = float(input("🌱 Enter land size (acres): "))
budget = float(input("💰 Enter your budget (₹): "))
transport = float(input("🚚 Enter transport cost (₹): "))

print("\n🤖 ANALYZING MARKET CONDITIONS...")
print("Please wait...\n")

results = {}

for crop in crops:
    results[crop] = calculate(
        crop,
        land,
        transport,
        budget
    )


# Sort crops by AI score
ranking = sorted(
    results.items(),
    key=lambda x: x[1]["score"],
    reverse=True
)


print("=" * 55)
print("              📊 CROP COMPARISON")
print("=" * 55)

    )


# Best crop
